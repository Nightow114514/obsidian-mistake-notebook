const kpRaw = await tp.system.prompt("考点（多个用英文逗号分隔）");
const reasonRaw = await tp.system.prompt("错因（多个用英文逗号分隔，如：概念不清,计算失误）");

  tR += "---\n";

  reasonList.forEach(r => { tR += `  - "${yamlEscape(r)}"\n`; });
  tR += `difficulty: "${diff}"\n`;
  tR += `answer: "${yamlEscape(ans)}"\n`;
  tR += "mastery: 未掌握\n";
  tR += `error-date: ${tp.date.now("YYYY-MM-DD")}\n`;
  tR += `next-review: ${tp.date.now("YYYY-MM-DD")}\n`;
  tR += "review-interval: 1\n";
  tR += "review-count: 0\n";
  tR += "---\n\n";
}
