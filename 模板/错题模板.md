<%*
const app = tp.app;
const wordPath = "脚本/规范词表.json";
let wordData = { "knowledge-point": [], "error-reason": [] };
try {
  const wordFile = app.vault.getAbstractFileByPath(wordPath);
  if (wordFile) {
    const raw = await app.vault.read(wordFile);
    if (raw) wordData = Object.assign(wordData, JSON.parse(raw));
  }
} catch (e) {
  new Notice("⚠️ 规范词表读取失败：" + e.message);
}

const uniqueSorted = (arr) =>
  Array.from(new Set((arr || []).map((s) => String(s).trim()).filter(Boolean)))
    .sort((a, b) => a.localeCompare(b, "zh-CN"));

const pickMany = async (key, label) => {
  const picked = [];
  const existing = uniqueSorted(wordData[key] || []);
  while (true) {
    const items = ["✨ 新建" + label, ...existing];
    const choice = await tp.system.suggester(items, items, false, "选择" + label + "（可直接打字过滤）");
    if (!choice) return null;
    let value;
    if (choice === "✨ 新建" + label) {
      value = await tp.system.prompt("输入新" + label);
      if (!value) return null;
      value = String(value).trim();
      if (!value) return null;
      if (!existing.includes(value)) existing.push(value);
    } else {
      value = choice;
    }
    if (!picked.includes(value)) picked.push(value);
    const more = await tp.system.suggester(
      ["✅ 完成", "➕ 继续添加" + label],
      [false, true],
      false,
      "还要继续添加" + label + "吗？"
    );
    if (more === null || more === undefined) break; // ESC = 结束收集，保留已选内容
    if (!more) break;
  }
  return { picked, existing: uniqueSorted(existing) };
};

const kpResult = await pickMany("knowledge-point", "考点");
const reasonResult = kpResult ? await pickMany("error-reason", "错因") : null;
const diff = reasonResult ? await tp.system.suggester(["简单", "中等", "困难"], ["简单", "中等", "困难"], false, "选择难度") : null;
const ans = diff ? ((await tp.system.prompt("正确答案（可留空，ESC 或直接确定=留空）")) ?? "") : null;

if (kpResult && reasonResult && diff) {
  wordData["knowledge-point"] = kpResult.existing;
  wordData["error-reason"] = reasonResult.existing;
  try {
    const wordFile = app.vault.getAbstractFileByPath(wordPath);
    const json = JSON.stringify(wordData, null, 2) + "\n";
    if (wordFile) await app.vault.modify(wordFile, json);
    else await app.vault.create(wordPath, json);
  } catch (e) {
    new Notice("⚠️ 规范词表保存失败：" + e.message);
  }

  const yamlEscape = (s) => String(s ?? "").replace(/\\/g, "\\\\").replace(/"/g, '\\"');
  tR += "---\n";
  tR += "type: 错题\n";
  tR += "knowledge-point:\n";
  kpResult.picked.forEach((k) => { tR += `  - "${yamlEscape(k)}"\n`; });
  tR += "error-reason:\n";
  reasonResult.picked.forEach((r) => { tR += `  - "${yamlEscape(r)}"\n`; });
  tR += `difficulty: "${yamlEscape(diff)}"\n`;
  tR += `answer: "${yamlEscape(ans)}"\n`;
  tR += "mastery: 未掌握\n";
  tR += `error-date: ${tp.date.now("YYYY-MM-DD")}\n`;
  tR += `next-review: ${tp.date.now("YYYY-MM-DD")}\n`;
  tR += "review-interval: 1\n";
  tR += "review-count: 0\n";
  tR += "mastery-streak: 0\n";
  tR += "review-log: []\n";
  tR += "---\n\n";
}
