# 🎲 随机组卷

> **使用方法**：切到编辑模式，修改下方 frontmatter 中的筛选条件，再切回阅读模式看结果。

| 字段 | 说明 | 示例 |
|---|---|---|
| `difficulty-filter` | 筛选难度 | `[简单, 中等]` 或 `[]`=不限 |
| `mastery-filter` | 筛选掌握程度 | `[未掌握, 部分掌握]` 或 `[]`=不限 |
| `kp-filter` | 考点筛选 | `三角函数` 或 `""`=不限 |
| `reason-filter` | 错因筛选 | `概念不清` 或 `""`=不限 |
| `due-only` | 卷子是否只含到期题 | `true`（默认）=只练到期题，不足 `count` 也不拿未到期题凑数；`false`=到期题优先，未到期题补满（加练） |
| `skip-recent-days` | 几天内抽过的不再抽 | `2`（默认）/ `0`=不限；**仅作用于未到期题**——已到期题永远优先出现、不受防重复限制 |
| `recent-days` | 只抽最近 N 天录入的题 | `0`=不限（默认）/ `3`=只抽最近 3 天新增 |
| `new-first` | 新题优先排序 | `false`=到期优先轮换（默认）/ `true`=按录入时间从新到旧（新题先抽满、老题补位；配合 `due-only: false` 可把昨天新增的题纳入今天练习） |
| `count` | 抽取数量 | `5` |
| `show-answer` | 答案是否直接显示 | `false`=屏幕隐藏防剧透（默认，导出 PDF 自动含答案）/ `true`=直接显示 |

## 📝 抽题结果

> 每道题下方的「✅ 会了 / ❌ 不会」按钮可直接更新该题的掌握程度与下次复习时间（复习次数 +1 并记入 `review-log`，见仪表盘「最近复习动态」）；「📝 写复盘」按钮可把复盘按「第 N 次复盘（日期）」追加到笔记的「我的复盘」区；「📖 打开笔记」按钮可直接跳转到对应笔记。**本套卷抽定后保持固定，标记不会换题**；点「✅ 本次已做完」才换新一批。**已到期题永远优先出现**（不受防重复限制，按最久没做轮换），标记「不会」的题 1 天后到期即会正常出现。答案默认隐藏，每题旁有「👀 查看答案」可单独展开；**导出 PDF 时答案会自动包含**，交互按钮自动隐藏。

const masteryLevels = dv.current()["mastery-filter"] || [];
const pickCount = Number(dv.current().count) || 5;
const targetKpRaw = dv.current()["kp-filter"] || "";
const targetReasonRaw = dv.current()["reason-filter"] || "";

const targetReasonList = targetReasonRaw.split(/[,，]/).map(s => s.trim()).filter(s => s.length > 0);

// 把日期值（Luxon / Date / 字符串）统一转成可比较的 Luxon，解析失败返回 null
const toLuxon = (v) => {
  if (v == null) return null;
  try {
    if (typeof v === "object" && typeof v.ts === "number" && typeof v.diff === "function") return v;
    return dv.date(String(v));
  } catch (e) { return null; }

}

function getCandidates() {
  let pages = dv.pages('"错题本"').where(p => p.type === "错题");

  if (difficulties.length > 0) {
    pages = pages.where(p => difficulties.includes(p.difficulty));
  }
  if (masteryLevels.length > 0) {
    pages = pages.where(p => masteryLevels.includes(p.mastery));
  }
  if (targetKpList.length > 0) {
    pages = pages.where(p => {
      if (!p["knowledge-point"]) return false;
      const kps = Array.isArray(p["knowledge-point"])
        ? p["knowledge-point"]
        : [p["knowledge-point"]];
      return targetKpList.some(tk => kps.some(k => k.includes(tk)));
    });
  }
  if (targetReasonList.length > 0) {
    pages = pages.where(p => {
      if (!p["error-reason"]) return false;
      const reasons = Array.isArray(p["error-reason"])
        ? p["error-reason"]
        : [p["error-reason"]];
      return targetReasonList.some(tr => reasons.some(r => r.includes(tr)));
    });
  }
  const isDue = (p) => !p["next-review"] || p["next-review"] <= today;
  if (recentDays > 0) {
    // 只抽最近 N 天录入的题（error-date >= 今天 - N）
    const cutoff = today.minus({ days: recentDays });
    pages = pages.where(p => {
      const d = toLuxon(p["error-date"]);
      return d ? d >= cutoff : false;
    });
  }
  if (dueOnly) {
    pages = pages.where(isDue);
  } else if (skipRecentDays > 0) {
    // 统一语义：已到期题永远优先、不受防重复限制（与 due-only 开关无关，
    // 与仪表盘「今日待复习」口径一致）；防重复窗口只作用于未到期题
    const cutoff = today.minus({ days: skipRecentDays });
    pages = pages.where(p => isDue(p) || !p["last-picked"] || p["last-picked"] < cutoff);
  }
  return Array.from(pages);
}

function shuffle(arr) {
  for (let i = arr.length - 1; i > 0; i--) {
    const j = Math.floor(Math.random() * (i + 1));
    [arr[i], arr[j]] = [arr[j], arr[i]];
  }
  return arr;
}

// 抽题：默认到期题优先且按「最久没做过」轮换（从未做过的最前，其次按 last-picked 从旧到新），
// 所有到期题过一遍前不会重复；未到期题跟在后面（随机）。due-only=true 时全部都是到期题。
// new-first=true 时改为按录入时间从新到旧优先（新题先抽满、老题补位，到期规则让位），
// 适合「昨天新增的题今天想再刷一遍」的场景
function drawPaths(candidates, n) {
  const isDue = (p) => !p["next-review"] || p["next-review"] <= today;
  const staleness = (a, b) => {
    const ap = a["last-picked"];
    const bp = b["last-picked"];
    if (!ap && !bp) return 0;
    if (!ap) return -1;
    if (!bp) return 1;
    const at = ap.toMillis ? ap.toMillis() : Date.parse(String(ap));
    const bt = bp.toMillis ? bp.toMillis() : Date.parse(String(bp));
    return at - bt;
  };
  const recency = (a, b) => {
    const at = toLuxon(a["error-date"]);
    const bt = toLuxon(b["error-date"]);
    if (!at && !bt) return 0;
    if (!at) return 1;   // 无录入日期排最后
    if (!bt) return -1;
    return bt.ts - at.ts; // 新 → 旧
  };
  if (newFirst) {
    const arr = shuffle(candidates);
    arr.sort(recency); // 稳定排序：同日期内保持随机序
    return arr.slice(0, n).map((p) => p.file.path);
  }
  const dueArr = shuffle(candidates.filter(isDue));
  dueArr.sort(staleness);
  const rest = shuffle(candidates.filter((p) => !isDue(p)));
  return dueArr.concat(rest).slice(0, n).map((p) => p.file.path);
}

const pad2 = (x) => String(x).padStart(2, "0");
const fmtDate = (d) => `${d.getFullYear()}-${pad2(d.getMonth() + 1)}-${pad2(d.getDate())}`;

// 掌握度递进规则：与 脚本/update-mastery.js 保持一致
// 未掌握 1 天；部分掌握 3/7/15/30 递进；已掌握 7/15/30 递进；
// 连续「已掌握」第 2 次 60 天、第 3 次起 90 天
function computeSchedule(page, direction) {
  const intervals = [1, 3, 7, 15, 30];
  const stepUp = (n) => {
    const i = intervals.indexOf(n);
    if (i === -1) return n >= 30 ? 30 : 3;
    return i < intervals.length - 1 ? intervals[i + 1] : 30;
  };
  const prevInt = Number(page["review-interval"]) || 1;
  const prevStreak = Number(page["mastery-streak"]) || 0;
  if (direction === "down") {
    return { mastery: "未掌握", interval: 1, streak: 0 };
  }
  const cur = page.mastery || "未掌握";
  if (cur === "已掌握") {
    const streak = prevStreak + 1;
    let interval = Math.max(7, stepUp(prevInt));
    if (streak >= 3) interval = 90;
    else if (streak >= 2) interval = 60;
    return { mastery: "已掌握", interval, streak };
  }
  if (cur === "部分掌握") {
    return { mastery: "已掌握", interval: Math.max(7, stepUp(prevInt)), streak: 0 };
  }
  return { mastery: "部分掌握", interval: Math.max(3, stepUp(prevInt)), streak: 0 };
}

// 往 frontmatter 追加一条 review-log（兼容块级列表 / 空行内数组 / 行内数组三种写法）
function appendLog(block, entry) {
  if (/^review-log:(\s*\[\])?\s*$/m.test(block)) {
    return block.replace(/^review-log:(\s*\[\])?\s*$/m, 'review-log:\n  - "' + entry + '"');
  }
  if (/^review-log:\s*\[[^\r\n]*\]\s*$/m.test(block)) {
    return block.replace(/^review-log:\s*\[([^\r\n]*)\]\s*$/m, (m0, inner) => {
      const sep = inner.trim() ? ", " : "";
      return 'review-log: [' + inner.trimEnd() + sep + '"' + entry + '"]';
    });
  }
  const idx = block.lastIndexOf("\n---");
  return block.slice(0, idx) + '\nreview-log:\n  - "' + entry + '"' + block.slice(idx);
}

// ---- 卷末批量落盘：卷内「会了/不会/写复盘」不立即写盘（避免每次点击触发 Dataview 重渲染），
// 攒到「最后一题标记完成 / 点已做完」时统一写入（只触发一次刷新）----
// 队列存 window（脚本重执行时保留）；重新进入页面时自动补落盘，防止中途退出丢失
function initPending(total, hash) {
  let p = window.__paperPending;
  if (!p || p.paperHash !== hash) {
    p = { paperHash: hash, total: total, baseDone: 0, marks: {}, refls: {} };
    window.__paperPending = p;
  }
  p.total = total;
  return p;
}

function getPending() {
  return window.__paperPending || null;
}

async function flushPending() {
  const pending = getPending();
  if (!pending || window.__paperFlushing) return;
  const hasWork = Object.keys(pending.marks).length > 0 || Object.keys(pending.refls).length > 0;
  if (!hasWork) return;
  window.__paperFlushing = true;
  try {
    // 按笔记聚合：每篇笔记一次 process 写入（掌握度字段 + review-log + 复盘条目）
    const noteOps = {};
    for (const [path, m] of Object.entries(pending.marks)) {
      (noteOps[path] = noteOps[path] || { marks: [], refls: [] }).marks.push(m);
    }
    for (const [path, rs] of Object.entries(pending.refls)) {
      (noteOps[path] = noteOps[path] || { marks: [], refls: [] }).refls.push(...rs);
    }
    for (const [path, op] of Object.entries(noteOps)) {
      const file = dv.app.vault.getAbstractFileByPath(path);
      if (!file || !file.extension) continue;
      await dv.app.vault.process(file, (data) => {
        const end = data.startsWith("---") ? data.indexOf("\n---", 3) : -1;
        let fm = end === -1 ? "" : data.slice(0, end + 4);
        let body = end === -1 ? data : data.slice(end + 4);
        const setProp = (block, key, value) => {
          const re = new RegExp("^" + key + ":.*$", "m");
          if (re.test(block)) return block.replace(re, key + ": " + value);
          const idx = block.lastIndexOf("\n---");
          return block.slice(0, idx) + "\n" + key + ": " + value + block.slice(idx);
        };
        for (const m of op.marks) {
          const countMatch = fm.match(/^review-count:\s*(\d+)/m);
          const newCount = countMatch ? Number(countMatch[1]) + 1 : 1;
          const logEntry = `${m.date} ${m.from}→${m.mastery}（间隔${m.interval}天）`;
          fm = fm.replace(/^mastery:.*$/m, "mastery: " + m.mastery);
          fm = setProp(fm, "next-review", m.nextReview);
          fm = setProp(fm, "review-interval", String(m.interval));
          fm = setProp(fm, "review-count", String(newCount));
          fm = setProp(fm, "mastery-streak", String(m.streak));
          fm = appendLog(fm, logEntry);
        }
        for (const r of op.refls) {
          const entry = `### 📝 第 ${r.n} 次复盘（${r.date}）\n${r.lines}\n`;
          const mm = body.match(/^## 📝 我的复盘[ \t]*$/m);
          if (mm) {
            const afterHeading = mm.index + mm[0].length;
            const rest = body.slice(afterHeading);
            const nextHeading = rest.match(/\n## /);
            const insertAt = nextHeading ? afterHeading + nextHeading.index : body.length;
            body = body.slice(0, insertAt).replace(/\s+$/, "") + "\n\n" + entry + body.slice(insertAt);
          } else {
            body = body.replace(/\s+$/, "") + "\n\n## 📝 我的复盘\n\n" + entry;
          }
        }
        return fm + body;
      });
    }
    // 标记结果一次写入组卷页 frontmatter
    if (Object.keys(pending.marks).length) {
      const results = { ...(dv.current()["paper-results"] || {}) };
      for (const [path, m] of Object.entries(pending.marks)) results[path] = m.direction;
      await setFrontmatter({ "paper-results": results });
    }
  } catch (e) {
    console.error("卷末批量落盘失败:", e);
  } finally {
    window.__paperPending = null;
    window.__paperFlushing = false;
  }
}

// 复盘：点「📝 写复盘」后展开内联输入框；保存进待落盘队列（卷末统一写入）

  wrap.style.margin = "0 0 12px 0";

  ta.style.width = "100%";
  ta.style.minHeight = "64px";
  const bar = wrap.createEl("div");
  bar.style.marginTop = "6px";
  const saveBtn = bar.createEl("button", { text: "💾 保存复盘" });

  statusEl.style.marginLeft = "10px";
  statusEl.style.fontSize = "0.9em";

function saveReflection(page, ta, statusEl, saveBtn, cancelBtn, reflLabel) {
  const text = ta.value.trim();
  if (!text) {
    statusEl.textContent = "⚠️ 复盘内容为空，未保存";
    return;
  }
  const pending = getPending() || initPending(0, dv.current()["paper-hash"] || "");
  const existingCount = (window.__paperReflCount && window.__paperReflCount[page.file.path]) || 0;
  const arr = pending.refls[page.file.path] || [];
  const n = existingCount + arr.length + 1;
  const lines = text.split(/\r?\n/).map((l) => "> " + l).join("\n");
  arr.push({ n, date: fmtDate(new Date()), lines });
  pending.refls[page.file.path] = arr;
  // 原地更新：移除输入区，显示保存状态（不触发页面刷新）
  ta.remove();
  saveBtn.remove();
  cancelBtn.remove();
  if (reflLabel) reflLabel.textContent = `已复盘 ${n} 次`;
  statusEl.textContent = `✅ 已记录第 ${n} 次复盘（卷末统一保存）`;
  statusEl.style.color = "var(--color-green)";
}

function markQuestion(page, direction, btnYes, btnNo, statusEl) {
  btnYes.disabled = true;
  btnNo.disabled = true;
  const sched = computeSchedule(page, direction);
  const now = new Date();
  const todayStr = fmtDate(now);
  const nextReview = fmtDate(new Date(now.getFullYear(), now.getMonth(), now.getDate() + sched.interval));
  // 写入待落盘队列（不写盘、不触发刷新）；最后一题标记完成后统一落盘
  const pending = getPending() || initPending(0, dv.current()["paper-hash"] || "");
  pending.marks[page.file.path] = {
    direction,
    mastery: sched.mastery,
    interval: sched.interval,
    streak: sched.streak,
    nextReview,
    date: todayStr,
    from: page.mastery || "未掌握"
  };
  if (direction === "up") {
    statusEl.textContent = `✅ 会了 → ${sched.mastery}，下次复习 ${nextReview}（卷末统一保存）`;
    statusEl.style.color = "var(--color-green)";
  } else {
    statusEl.textContent = "❌ 不会 → 未掌握，1 天后复习（卷末统一保存）";
    statusEl.style.color = "var(--color-red)";
  }
  // 全部题目都已标记 → 统一落盘（只触发一次刷新）
  const doneCount = (pending.baseDone || 0) + Object.keys(pending.marks).length;
  if ((pending.total || 0) > 0 && doneCount >= pending.total) {
    flushPending();
  }
}

async function renderPaper() {
  dv.container.empty();

  const cur = dv.current();
  const storedHash = cur["paper-hash"] || "";
  const storedPaths = Array.isArray(cur["paper-questions"]) ? cur["paper-questions"] : [];
  let results = cur["paper-results"] || {};

  let paths;
  if (storedHash === filtersHash() && storedPaths.length > 0) {
    // 恢复当前卷：标记会了/不会、Dataview 自动刷新、切换笔记都不会换题
    paths = storedPaths;
  } else {
    // 换卷前先落盘上一卷的待定改动，防止丢失
    await flushPending();
    // 抽一套新卷并持久化
    const candidates = getCandidates();
    if (candidates.length === 0) {
      dv.paragraph("⚠️ 没有符合条件的错题，请放宽筛选条件。");
      return;
    }
    const n = Math.min(pickCount, candidates.length);
    paths = drawPaths(candidates, n);
    results = {};
    setFrontmatter({ "paper-hash": filtersHash(), "paper-questions": paths, "paper-results": {} });
  }

  const pages = paths.map((p) => dv.page(p)).filter(Boolean);
  if (pages.length === 0) {
    dv.paragraph("⚠️ 本套卷题目已全部删除，点下方按钮重新抽卷。");
    const resetBtn = dv.el("button", "🔄 重新抽卷");
    resetBtn.classList.add("paper-buttons");
    resetBtn.addEventListener("click", () => {
      setFrontmatter({ "paper-hash": "", "paper-questions": [], "paper-results": {} });
    });
    return;
  }

  // 待落盘队列初始化：total=本卷题数，baseDone=已落盘标记数（组卷页 frontmatter 的 paper-results）
  const pending = initPending(pages.length, storedHash);
  pending.baseDone = Object.keys(results).length;

  // 重新进入页面且存在未落盘改动 → 立即补落盘（写盘会触发重渲染，下一轮渲染显示最终状态）
  if (!window.__paperFlushing &&
      (Object.keys(pending.marks).length > 0 || Object.keys(pending.refls).length > 0)) {
    flushPending();
    return;
  }

  // 读取各笔记的复盘次数，渲染时显示「已复盘 N 次」
  const reflCount = {};
  for (const p of pages) {
    try {
      const f = dv.app.vault.getAbstractFileByPath(p.file.path);
      if (f) {
        const c = await dv.app.vault.read(f);
        reflCount[p.file.path] = (c.match(/^### 📝 第 \d+ 次复盘/mg) || []).length;
      }
    } catch (e) { /* 忽略读取失败 */ }
  }
  window.__paperReflCount = reflCount;

  const markedCount = pages.filter((p) => results[p.file.path]).length;
  dv.header(2, `本套卷：${pages.length} 道（已标记 ${markedCount}/${pages.length}）`);

  dv.paragraph("---");

  pages.forEach((p, idx) => {
    const kps = Array.isArray(p["knowledge-point"])
      ? p["knowledge-point"].join(" · ")
      : p["knowledge-point"] || "-";
    dv.paragraph(`**第 ${idx + 1} 题**　｜　${p.difficulty || "-"}　｜　${kps}`);

    if (showAnswer) {
      dv.paragraph(`答案：${p.answer || "-"}`);
    } else {
      // 每题一个「查看答案」按钮：单独展开/收起本题答案（paper-answer-show / paper-answer-hide 类）；
      // 导出 PDF 时由 @media print 强制显示全部答案
      const ansRow = dv.container.createEl("div");
      ansRow.style.margin = "2px 0";
      const ansBtn = ansRow.createEl("button", { text: "👀 查看答案" });
      ansBtn.classList.add("paper-buttons");
      ansBtn.style.fontSize = "0.85em";
      const ansEl = ansRow.createEl("div", { cls: "paper-answer" });
      ansEl.style.margin = "4px 0";
      ansEl.style.paddingLeft = "18px";
      ansEl.textContent = `答案：${p.answer || "-"}`;
      ansBtn.addEventListener("click", () => {
        if (ansEl.classList.contains("paper-answer-show")) {
          ansEl.classList.remove("paper-answer-show");
          ansEl.classList.add("paper-answer-hide");
          ansBtn.textContent = "👀 查看答案";
        } else {
          ansEl.classList.remove("paper-answer-hide");
          ansEl.classList.add("paper-answer-show");
          ansBtn.textContent = "🙈 隐藏答案";
        }
      });
    }

    // 显示笔记里的全部截图
    const imageLinks = (p.file.outlinks || []).filter((l) =>
      /\.(jpe?g|png|gif|webp|bmp|svg)$/i.test(l.path)
    );
    if (imageLinks.length > 0) {
      imageLinks.forEach((link) => dv.paragraph(`![[${link.path}]]`));
    } else {
      dv.paragraph(`![[${p.file.name}]]`);
    }

    // 已标记 → 显示结果；未标记 → 会了 / 不会按钮
    const res = results[p.file.path];
    if (res) {
      const doneEl = dv.container.createEl("div", { cls: "paper-buttons" });
      doneEl.style.margin = "6px 0 12px 0";
      doneEl.style.fontSize = "0.95em";
      // 下次复习日期直接读笔记最新 frontmatter（刷新后仍常驻显示）
      const nr = p["next-review"];
      const nextStr = nr
        ? (typeof nr.toISODate === "function" ? nr.toISODate() : String(nr).slice(0, 10))
        : "尽快";
      if (res === "up") {
        doneEl.textContent = `✅ 已标记：会了 → ${p.mastery || "未掌握"}，下次复习 ${nextStr}`;
        doneEl.style.color = "var(--color-green)";
      } else {
        doneEl.textContent = `❌ 已标记：不会 → 未掌握，下次复习 ${nextStr}`;
        doneEl.style.color = "var(--color-red)";
      }
    } else {
      const btnRow = dv.container.createEl("div", { cls: "paper-buttons" });
      btnRow.style.margin = "6px 0 12px 0";
      const btnYes = btnRow.createEl("button", { text: "✅ 会了" });
      const btnNo = btnRow.createEl("button", { text: "❌ 不会" });
      btnNo.style.marginLeft = "8px";
      const statusEl = btnRow.createEl("span");
      statusEl.style.marginLeft = "10px";
      statusEl.style.fontSize = "0.9em";
      btnYes.addEventListener("click", () => markQuestion(p, "up", btnYes, btnNo, statusEl));
      btnNo.addEventListener("click", () => markQuestion(p, "down", btnYes, btnNo, statusEl));
    }

    // 复盘与跳转（已标记/未标记的题都可用）
    const utilRow = dv.container.createEl("div", { cls: "paper-buttons" });
    utilRow.style.margin = "0 0 12px 0";
    const btnRefl = utilRow.createEl("button", { text: "📝 写复盘" });
    const btnOpen = utilRow.createEl("button", { text: "📖 打开笔记" });
    btnOpen.style.marginLeft = "8px";
    const reflLabel = utilRow.createEl("span");
    reflLabel.style.marginLeft = "10px";
    reflLabel.style.fontSize = "0.85em";
    reflLabel.style.color = "var(--text-muted)";
    reflLabel.textContent = `已复盘 ${reflCount[p.file.path] || 0} 次`;
    btnOpen.addEventListener("click", () => {
      dv.app.workspace.openLinkText(p.file.path, "", false);
    });
    btnRefl.addEventListener("click", () => {
      btnRefl.disabled = true;
      const editor = createReflEditor(p, reflLabel);
      utilRow.after(editor);
    });

    dv.paragraph("---");
  });

  const markBtn = dv.el("button", `✅ 本次已做完：标记这 ${pages.length} 道题并换下一批`);
  markBtn.classList.add("paper-buttons");
  markBtn.style.margin = "12px 0 4px 0";

  // 换批规则说明（打印时随按钮一起隐藏）
  const ruleHint = dv.container.createEl("div");
  ruleHint.classList.add("paper-buttons");
  ruleHint.style.margin = "0 0 12px 0";
  ruleHint.style.fontSize = "0.85em";
  ruleHint.style.color = "var(--text-muted)";
  ruleHint.textContent = dueOnly
    ? "换批规则：标记过「会了 / 不会」的题按各自复习间隔到期后再出现；还没标记的到期题会继续轮换出现。"
    : `换批规则：已到期题优先出现、不受防重复限制；未到期题${skipRecentDays > 0 ? " " + skipRecentDays + " 天内不重复" : "不限制"}。`;
  markBtn.addEventListener("click", async () => {
    markBtn.disabled = true;
    markBtn.textContent = "正在标记…";
    // 先统一落盘卷内的待定改动（会了/不会/写复盘）
    await flushPending();
    const now = new Date();
    const todayStr = fmtDate(now);
    for (const p of pages) {
      const file = dv.app.vault.getAbstractFileByPath(p.file.path);
      if (!file || !file.extension) continue;
      try {
        await dv.app.vault.process(file, (data) => {
          if (!data.startsWith("---")) return data;
          const end = data.indexOf("\n---", 3);
          if (end === -1) return data;
          const fm = data.slice(0, end + 4);
          if (/^last-picked:.*$/m.test(fm)) {
            return fm.replace(/^last-picked:.*$/m, "last-picked: " + todayStr) + data.slice(end + 4);
          }
          const idx = fm.lastIndexOf("\n---");
          return fm.slice(0, idx) + "\nlast-picked: " + todayStr + fm.slice(idx) + data.slice(end + 4);
        });
      } catch (e) {
        console.error("标记 last-picked 失败:", e);
      }
    }
    // 清空本卷会话 → Dataview 刷新后自动抽新一批
    await setFrontmatter({ "paper-hash": "", "paper-questions": [], "paper-results": {} });
  });
}

renderPaper();
