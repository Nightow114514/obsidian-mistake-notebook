# 📊 错题仪表盘

## 🗓️ 今日待复习

```dataviewjs
const today = dv.date("today");
const pages = dv.pages('"错题本"')
  .where(p => p.type === "错题" && (!p["next-review"] || p["next-review"] <= today));

if (pages.length === 0) {
  dv.paragraph("🎉 今日没有到期错题，全部搞定！");
} else {
  const fmtDate = (d) => d && d.toISODate ? d.toISODate() : (d ? String(d).slice(0, 10) : "尽快");
  dv.header(3, `今日待复习：${pages.length} 道`);
  dv.table(
    ["题目", "难度", "考点", "错因", "已复习", "下次复习"],
    pages.sort(p => p["next-review"] || today, "asc").map(p => [
      p.file.link,
      p.difficulty || "-",
      (Array.isArray(p["knowledge-point"]) ? p["knowledge-point"].join(" · ") : p["knowledge-point"]) || "-",
      (Array.isArray(p["error-reason"]) ? p["error-reason"].join(" · ") : p["error-reason"]) || "-",
      p["review-count"] || 0,
      fmtDate(p["next-review"])
    ])
  );
}
```

## ⚠️ 薄弱考点 TOP 10

```dataview
TABLE 
  length(rows) AS "错题数"
FROM "错题本"
WHERE type = "错题"
FLATTEN knowledge-point AS kp
GROUP BY kp
SORT length(rows) DESC
LIMIT 10
```

## 💡 错因分布

```dataview
TABLE 
  length(rows) AS "错题数"
FROM "错题本"
WHERE type = "错题" AND error-reason
FLATTEN error-reason AS reason
GROUP BY reason
SORT length(rows) DESC
```

## 📈 掌握程度分布图

```dataviewjs
const pages = dv.pages('"错题本"').where(p => p.type === "错题");
const total = pages.length;
if (total === 0) {
  dv.paragraph("暂无数据，快去录错题吧 ✍️");
} else {
  const counts = { "未掌握": 0, "部分掌握": 0, "已掌握": 0 };
  for (const p of pages) {
    const m = p.mastery || "未掌握";
    if (counts[m] !== undefined) counts[m]++;
  }

  const colors = { "未掌握": "#e74c3c", "部分掌握": "#f39c12", "已掌握": "#27ae60" };

  for (const [label, count] of Object.entries(counts)) {
    const pct = ((count / total) * 100).toFixed(0);
    const bar = "█".repeat(Math.round(count / total * 30));
    dv.paragraph(`**${label}**　${count} 题（${pct}%）`);
    dv.el("div", bar, { style: `color:${colors[label]}; font-size:14px; margin-bottom:8px;` });
  }
}
```

## 📈 每周新增趋势

```dataviewjs
const pages = dv.pages('"错题本"').where(p => p.type === "错题" && p["error-date"]);
if (pages.length === 0) {
  dv.paragraph("暂无数据");
} else {
  // 按周汇总
  const weeks = {};
  for (const p of pages) {
    const d = new Date(p["error-date"]);
    if (isNaN(d.getTime())) continue;
    // 计算该日期所在周的周一
    const day = d.getDay() || 7;
    const monday = new Date(d);
    monday.setDate(d.getDate() - day + 1);
    const key = monday.toISOString().slice(0, 10);
    weeks[key] = (weeks[key] || 0) + 1;
  }

  const sorted = Object.entries(weeks).sort();
  const maxCount = Math.max(...sorted.map(([,c]) => c), 1);

  dv.table(
    ["周", "新增", "趋势"],
    sorted.map(([week, count]) => {
      const barLen = Math.round(count / maxCount * 20);
      return [
        week,
        count,
        "▬".repeat(barLen) + " " + count
      ];
    })
  );
}
```
